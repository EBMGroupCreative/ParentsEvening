function shRound(ctx, x, y, w, h, r) { ctx.beginPath(); ctx.moveTo(x + r, y); ctx.arcTo(x + w, y, x + w, y + h, r); ctx.arcTo(x + w, y + h, x, y + h, r); ctx.arcTo(x, y + h, x, y, r); ctx.arcTo(x, y, x + w, y, r); ctx.closePath(); }
function shMono(ctx, t, x, y, sz, col, al) { ctx.font = '500 ' + sz + 'px ' + SH.M; ctx.fillStyle = col || SH.INK; ctx.textAlign = al || 'left'; ctx.letterSpacing = '2px'; ctx.fillText(String(t).toUpperCase(), x, y); ctx.letterSpacing = '0px'; ctx.textAlign = 'left'; }
function shNoWidow(out) { const n = out.length; if (n > 1 && !/\s/.test(out[n - 1].trim())) { const p = out[n - 2].split(' '); if (p.length > 2) { out[n - 1] = p.pop() + ' ' + out[n - 1]; out[n - 2] = p.join(' '); } } return out; }
function shSel(ctx, text, x, y, size) {
  ctx.font = '800 ' + size + 'px ' + SH.G; ctx.letterSpacing = Math.round(-size * 0.04) + 'px';
  const w = ctx.measureText(text).width, top = y - size * 0.8, h = size * 0.98, hw = Math.max(3, size * 0.026), k = size * 0.075;
  ctx.fillStyle = SH.HL; ctx.fillRect(x - size * 0.04, top, w + size * 0.08, h);
  ctx.fillStyle = SH.BLUE; ctx.fillRect(x - size * 0.05, top, hw, h); ctx.fillRect(x + w + size * 0.03, top, hw, h);
  ctx.beginPath(); ctx.arc(x - size * 0.05 + hw / 2, top - k * 0.6, k, 0, 7); ctx.fill();
  ctx.beginPath(); ctx.arc(x + w + size * 0.03 + hw / 2, top + h + k * 0.6, k, 0, 7); ctx.fill();
  ctx.fillStyle = SH.INK; ctx.fillText(text, x, y); ctx.letterSpacing = '0px';
  return w;
}

function useShare(ref, file, text) {
  const [note, setNote] = React.useState(null);
  const blob = () => new Promise(res => ref.current.toBlob(res, 'image/png'));
  const save = async () => { const b = await blob(); const a = document.createElement('a'); a.href = URL.createObjectURL(b); a.download = file; a.click(); setNote('Saved. Now send it to someone who deserves it.'); };
  const canShare = !!(navigator.canShare && navigator.canShare({ files: [new File([''], 'x.png', { type: 'image/png' })] }));
  const share = async () => { const b = await blob(); try { await navigator.share({ files: [new File([b], file, { type: 'image/png' })], text }); } catch (e) {} };
  return { note, setNote, save, share, canShare };
}

// Rich-text wrap for canvas: tokens [{s, hl}] → lines of tokens.
function shWrapRich(ctx, toks, maxW) {
  // group pieces into words; pieces not separated by whitespace stay glued (e.g. 'Ekse' + '.' + '”')
  const words = []; let glue = false;
  toks.forEach(tk => String(tk.s).split(/(\s+)/).forEach(p => {
    if (!p) return;
    if (/^\s+$/.test(p)) { glue = false; return; }
    if (glue && words.length) words[words.length - 1].push({ s: p, hl: tk.hl }); else words.push([{ s: p, hl: tk.hl }]);
    glue = true;
  }));
  const sp = ctx.measureText(' ').width, lines = [[]]; let w = 0;
  words.forEach(wd => {
    const ww = wd.reduce((s, p) => s + ctx.measureText(p.s).width, 0);
    const cur = lines[lines.length - 1];
    if (cur.length && w + sp + ww > maxW) { lines.push([...wd]); w = ww; return; }
    if (cur.length) { cur.push({ s: ' ', hl: false }); w += sp; }
    cur.push(...wd); w += ww;
  });
  const ls = lines.filter(l => l.length), n = ls.length;
  // no widows: if the last line is one word, pull the previous line's last word down
  if (n > 1 && !ls[n - 1].some(t => t.s === ' ')) { const prev = ls[n - 2], k = prev.map(t => t.s).lastIndexOf(' '); if (k > 0 && prev.slice(0, k).some(t => t.s === ' ')) { ls[n - 1] = [...prev.slice(k + 1), { s: ' ', hl: false }, ...ls[n - 1]]; ls[n - 2] = prev.slice(0, k); } }
  return ls;
}
