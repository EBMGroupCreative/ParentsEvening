function rr(ctx, x, y, w, h, r) { ctx.beginPath(); ctx.moveTo(x + r, y); ctx.arcTo(x + w, y, x + w, y + h, r); ctx.arcTo(x + w, y + h, x, y + h, r); ctx.arcTo(x, y + h, x, y, r); ctx.arcTo(x, y, x + w, y, r); ctx.closePath(); }

function dither2(ctx, img, x, y, w, h, { cell, style = 'Halftone', zoom, posX = 0.5, posY = 0.3 }) {
  const cols = Math.floor(w / cell), rows = Math.floor(h / cell);
  const off = document.createElement('canvas'); off.width = cols; off.height = rows;
  const o = off.getContext('2d');
  const sc = Math.max(cols / img.width, rows / img.height) * zoom;
  o.fillStyle = '#fff'; o.fillRect(0, 0, cols, rows);
  o.drawImage(img, (cols - img.width * sc) * posX, (rows - img.height * sc) * posY, img.width * sc, img.height * sc);
  const d = o.getImageData(0, 0, cols, rows).data; const lum = new Float32Array(cols * rows); let mn = 1, mx = 0;
  for (let i = 0; i < cols * rows; i++) { const l = (0.299 * d[i * 4] + 0.587 * d[i * 4 + 1] + 0.114 * d[i * 4 + 2]) / 255; lum[i] = l; mn = Math.min(mn, l); mx = Math.max(mx, l); }
  const span = Math.max(0.2, mx - mn); ctx.fillStyle = '#111';
  for (let r = 0; r < rows; r++) for (let c = 0; c < cols; c++) {
    let l = (lum[r * cols + c] - mn) / span; l = Math.min(1, Math.max(0, (l - 0.5) * 1.25 + 0.5));
    const px = x + c * cell, py = y + r * cell;
    if (style === 'Halftone') { const rad = cell * 0.66 * Math.sqrt(1 - l); if (rad > 0.4) { ctx.beginPath(); ctx.arc(px + cell / 2, py + cell / 2, rad, 0, Math.PI * 2); ctx.fill(); } }
    else if (l < (BAYER2[r % 4][c % 4] + 0.5) / 16) ctx.fillRect(px, py, cell, cell);
  }
}

              <label className="pos-r"><span className="lbl muted">Up</span><input type="range" min="0" max="1" step="0.01" value={1 - posY} disabled={!slackY} onChange={e => setPosY(1 - e.target.value)} /><span className="lbl muted">Down</span></label>
            </div>
            {(!slackX || !slackY) && <span className="lbl muted">{'Zoom in to move it ' + (!slackX && !slackY ? 'either way' : !slackX ? 'left or right' : 'up or down') + '. The frame is already full.'}</span>}
          </div>}
          <div className="ctl-row">
            <div className="ctl"><span className="lbl muted">Treatment</span><div className="row" style={{ gap: 6 }}>{['Pixelate', 'Halftone'].map(s => <button key={s} type="button" className={'chip' + (style === s ? ' on' : '')} onClick={() => setStyle(s)}>{s}</button>)}</div></div>
            <div className="ctl"><span className="lbl muted">{style + ' scale'}</span><div className="row" style={{ gap: 6 }}>{[['Fine', 4], ['Medium', 8], ['Chunky', 13], ['Heavy', 18], ['Billboard', 26]].map(([s, v]) => <button key={s} type="button" className={'chip' + (cell === v ? ' on' : '')} onClick={() => setCell(v)}>{s}</button>)}</div></div>
          </div>
          <label className="ctl"><span className="lbl muted">Name on the card</span><input ref={nameRef} className="field" value={name} onChange={e => setName(e.target.value)} placeholder="Your name" /><BlockNote v={name} msg={CARD_BLOCKED} /></label>
          {me && <label className="ctl"><span className="lbl muted">Origin · as written at enrolment, change it if you like</span><input className="field" value={originV} onChange={e => setOrigin(e.target.value)} placeholder="e.g. JHB, GP" maxLength={24} autoComplete="off" autoCorrect="off" autoCapitalize="off" spellCheck={false} /><span className="between"><span><BlockNote v={originV} msg={CARD_BLOCKED} /></span><span className="lbl muted">{originV.length + ' / 24 · prints in capitals'}</span></span></label>}
          <div ref={titleRef} className="ctl"><span className="lbl muted">Title</span>
            <div className="tp-groups">
              {TITLE_GROUPS.map(([g, list]) => { const open = tg === g, has = list.includes(level); return (
                <div key={g} className={'tp-grp' + (open ? ' open' : '')}>
                  <button type="button" className="tp-head" aria-expanded={open} onClick={() => setTg(open ? null : g)}><span className="lbl">{g}</span><span className="lbl muted">{has ? level.n : open ? 'Close' : list.length + ' titles'}</span></button>
                  {open && <div className="tp-list" role="listbox" aria-label={g}>{list.map(t => <button key={t.n} type="button" role="option" aria-selected={t === level} className={'tp-o' + (t === level ? ' on' : '')} onClick={() => { setTitleN(t.n); setTg(null); }}><b>{t.n}</b><span>{t.d}</span></button>)}</div>}
                </div>
              ); })}
            </div>
          </div>
          <label className="ctl"><span className="lbl muted">Favourite word</span>
            <select ref={wordRef} className="field" value={word} onChange={e => setWord(e.target.value)}>{window.LSD2.lessons.map(l => <option key={l.n} value={l.w}>{l.n + ' · ' + l.w}</option>)}</select>
          </label>
          <div className="row">
            <button type="button" className="btn primary" onClick={save}>[ Download card ]</button>
            <button type="button" className={'btn sm ' + (sheet ? 'soft' : 'line')} onClick={share}>Share to socials</button>
            <button type="button" className="btn line sm" onClick={copy}>Copy caption</button>
          </div>
          {sheet && <div className="card share-sheet">
            <div className="between lbl"><span className="blue">Share to</span><button type="button" className="chip" style={{ padding: '6px 12px', fontSize: 13 }} onClick={() => setSheet(false)}>Close</button></div>
            <div className="row" style={{ gap: 8 }}>{socials.map(([k, u]) => <a key={k} className="btn line sm" href={u} target="_blank" rel="noopener">{k}</a>)}<button type="button" className="btn line sm" onClick={save}>Instagram story</button></div>
            <span className="lbl muted">Desktop browsers won’t attach the picture for you. Download the card, then attach it. Instagram: download, open the app, post to your story. On a phone, Share opens your apps directly.</span>
          </div>}
          {note && <div className="lbl blue">{note}</div>}
          <a className="btn primary" href="#lessons/01" style={{ alignSelf: 'flex-start' }}>{'Card sorted. On to Lesson 01 →'}</a>
        </div>
        <div className={'idc-prev' + (claimed ? ' docked' : '')}><div className="idc-cardcol"><div className={'idc-drop' + (drag ? ' on' : '')} onDragOver={e => { e.preventDefault(); setDrag(true); }} onDragLeave={() => setDrag(false)} onDrop={e => { e.preventDefault(); setDrag(false); load(e.dataTransfer.files[0]); }}><canvas ref={cv} width={CARD_W2} height={CARD_H2}></canvas>{HOT.map(([k, lab, x, y, w, h, go]) => <button key={k} type="button" className={'idc-hot ' + k} aria-label={lab} title={lab} style={{ left: (x / CARD_W2 * 100) + '%', top: (y / CARD_H2 * 100) + '%', width: (w / CARD_W2 * 100) + '%', height: (h / CARD_H2 * 100) + '%' }} onClick={go}><span className="lbl">{k === 'photo' && !img ? 'Drop or tap to add a photo' : lab}</span></button>)}</div><div className="lbl muted" style={{ textAlign: 'center' }}>Tap the photo, name, title or word on the card to change it</div>{me && !claimed && <button type="button" className="btn primary vc-claim" onClick={claim}>Done. Claim my Study Grant.</button>}<a className="lbl muted" href="#report" style={{ textAlign: 'center' }}>Report a card</a></div>{claimed && <VoucherDock vouchers={vouchers} setVouchers={setVouchers} no={no} freshId={freshV} />}</div>
      </div>
    </div>
  );
}
